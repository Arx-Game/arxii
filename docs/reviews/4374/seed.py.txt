def answer(tier, *, pledge_row=pledge, **parts):
    label = tier.lower()
    fields = {
        "being": being,
        "pledge": pledge_row,
        "outcome_tier": tiers[tier],
        "delay_text": f"[staff-authored delay text: {label}]",
        "answer_text": f"[staff-authored answer text: {label}]",
        "cost_text": f"[staff-authored cost text: {label}]",
        "cost_pool": pool,
        "onlooker_line": f"[staff-authored onlooker line: {label}]",
    }
    fields.update(parts)
    row = AnswerFactory(**fields)
    AnswerSacrificeLineFactory(answer=row, facet=silk, text=f"[staff-authored sacrifice line: Silk, {label}]")
    return row


# Success is authored through the Answer builder during the pass. Partial and lower:

def pc(name, *, chosen=False, rollmod=0):
    character = create_object(
        "typeclasses.characters.Character", key=name, location=shrine_room, home=shrine_room
    )
    sheet = CharacterSheetFactory(character=character)
    if rollmod:
        sheet.rollmod = rollmod
        sheet.save(update_fields=["rollmod"])
    entry_row = RosterEntryFactory(character_sheet=sheet)
    RosterTenureFactory(roster_entry=entry_row, player_data=account.player_data, end_date=None)
    ActionPointPoolFactory(character=sheet, current=50, maximum=50)
    SceneParticipation.objects.get_or_create(scene=scene, account=account)
    if chosen:
        DevotionStandingFactory(
            character_sheet=sheet,
            being=being,
            valence=PatronageValence.DEVOTIONAL,
            established_at=timezone.now(),
        )
    return character, sheet


def carried(character, sheet, name, value, facet=None):
    obj = create_object("typeclasses.objects.Object", key=name, location=character)
    item = ItemInstanceFactory(
        template__name=name, template__value=value, game_object=obj, holder_character_sheet=sheet
    )
    if facet is not None:
        ItemFacetFactory(item_instance=item, facet=facet)
    return item


people = {}
for name, chosen, rollmod, items in (

):

    for item_name, value, facet in items:
        carried(character, sheet, item_name, value, facet)
