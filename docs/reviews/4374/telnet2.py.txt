"""Two real telnet sessions on :4000: an onlooker who stays connected in the room, and an
actor who runs commands. Prints what each one received."""
import json, re, socket, sys, time

def clean(buf):
    text = re.sub(rb"\xff[\xfb-\xfe].|\xff\xfa.*?\xff\xf0|\xff.", b"", buf, flags=re.S)
    return re.sub(r"\x1b\[[0-9;]*m", "", text.decode("utf-8", "replace"))

class Session:
    def __init__(self, user, char):
        self.s = socket.create_connection(("localhost", 4000), timeout=5)
        self.read(2)
        print(f"[{char}] connect: " + self.send(f"connect {user} TestPass123!", log=False).strip()[-300:])
        print(f"[{char}] @ic: " + self.send(f"@ic {char}").strip()[-400:])
    def read(self, wait=1.5):
        end = time.time() + wait; buf = b""
        while time.time() < end:
            try:
                self.s.settimeout(0.3); chunk = self.s.recv(65536)
                if not chunk: break
                buf += chunk
            except socket.timeout:
                pass
        return clean(buf)
    def send(self, cmd, log=True, wait=3.0):
        self.s.sendall((cmd + "\r\n").encode())
        out = self.read(wait)
        return out
    def close(self):
        self.s.sendall(b"quit\r\n"); self.s.close()

plan = json.loads(sys.argv[1])
watcher = Session(plan.get("onlooker_user", plan["user"]), plan["onlooker"])
watcher.read(1)
actor = Session(plan["user"], plan["actor"])
print(f"===== actor {plan['actor']}")
for cmd in plan["commands"]:
    print(f"> {cmd}")
    print(actor.send(cmd, wait=4.0).strip())
time.sleep(2)
print(f"===== onlooker {plan['onlooker']} received while connected")
print(watcher.read(3).strip())
actor.close(); watcher.close()
