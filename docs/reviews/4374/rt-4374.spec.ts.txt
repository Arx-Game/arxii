async function login(page: Page, user: string, password: string) {
  await page.goto(`${BASE}/login`);
  await page.waitForLoadState('networkidle');
  await page.getByPlaceholder('Username or Email').fill(user);
  await page.getByPlaceholder('Password').fill(password);
  await page.getByRole('button', { name: 'Log In', exact: true }).click();
  await page.waitForURL((url) => !url.pathname.startsWith('/login'), { timeout: 15000 });
}

async function adminLogin(page: Page) {
  await login(page, 'staff_4374', 'StaffPass123!');
}

async function csrf(page: Page): Promise<string> {
  const cookies = await page.context().cookies();
  return cookies.find((c) => c.name === 'csrftoken')?.value ?? '';
}

async function dispatch(page: Page, character: number, key: string, kwargs: object) {
  const res = await page.request.post(`${BASE}/api/actions/characters/${character}/dispatch/`, {
    headers: { 'X-CSRFToken': await csrf(page), Referer: BASE },
    data: { ref: { backend: 'registry', registry_key: key }, kwargs },
  });
  return { status: res.status(), body: await res.json() };
}

async function visions(page: Page, sheet: number) {
  const res = await page.request.get(`${BASE}/api/worship/visions/?recipient=${sheet}`);
  return (await res.json()).results;
}

async function pick(page: Page, field: string, text: string) {
  await page.locator(`[data-field="${field}"] .select2-selection`).click();
  await page.locator('.select2-search__field').fill(text);
  await page.locator('.select2-results__option', { hasText: text }).first().click();
}
