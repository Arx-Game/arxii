"""Round 1 setup (server stopped). Item 4: a second pledge type whose Failure and Critical
Failure answers carry fact text and a fact clue saved straight to the database, skipping
Answer.clean, and Ivo (rollmod -100) pledged to it. Item 8: the held-answer task's last run
moved a day back so the running server's scheduler fires it."""
from datetime import timedelta

from django.utils import timezone
from evennia import create_object
from evennia.accounts.models import AccountDB
from evennia.objects.models import ObjectDB

from world.action_points.factories import ActionPointPoolFactory
from world.ceremonies.services import sacrifice_items
from world.character_sheets.factories import CharacterSheetFactory
from world.clues.models import Clue
from world.game_clock.models import ScheduledTaskRecord
from world.items.factories import ItemInstanceFactory
from world.roster.factories import RosterEntryFactory, RosterTenureFactory
from world.traits.models import CheckOutcome
from world.worship.models import Answer, Pledge, PendingAnswer, Vision, WorshippedBeing
from world.worship.pledge_services import take_pledge

being = WorshippedBeing.objects.get(name="Ombrin")
p1 = Pledge.objects.get(pk=1)
p2 = Pledge.objects.create(being=being, name="[staff-authored pledge name 2]", description="[staff-authored pledge description 2]", favor_target=5, span_weeks=2, closing_rite=p1.closing_rite)
fact_clue = Clue.objects.get(name="Ombrin codex clue")
for tier in ("Failure", "Critical Failure"):
    row = Answer(being=being, pledge=p2, outcome_tier=CheckOutcome.objects.get(name=tier),
                 delay_text=f"[staff-authored delay text: {tier.lower()}, pledge 2]",
                 answer_text=f"[staff-authored answer text: {tier.lower()}, pledge 2]",
                 fact_text=f"[FACT THAT MUST NOT BE SENT: {tier.lower()}]", fact_clue=fact_clue,
                 onlooker_line=f"[staff-authored onlooker line: {tier.lower()}, pledge 2]")
    row.save()  # straight to the database: Answer.clean (which refuses this) never runs
    print("SAVED_ANSWER", row.pk, tier, "fact_text set:", bool(row.fact_text), "fact_clue:", row.fact_clue_id)

account = AccountDB.objects.get(username="e2e_test_account")
room = ObjectDB.objects.get(db_key="Ombrin's Shrine")
ivo = create_object("typeclasses.characters.Character", key="Ivo Marsh", location=room, home=room)
sheet = CharacterSheetFactory(character=ivo)
sheet.rollmod = -100
sheet.save(update_fields=["rollmod"])
entry = RosterEntryFactory(character_sheet=sheet)
RosterTenureFactory(roster_entry=entry, player_data=account.player_data, end_date=None)
ActionPointPoolFactory(character=sheet, current=50, maximum=50)
ivo.locks.add(f"puppet:pid({account.id}) or perm(Developer) or pperm(Developer)")
obj = create_object("typeclasses.objects.Object", key="iron nail", location=ivo)
item = ItemInstanceFactory(template__name="iron nail", template__value=50, game_object=obj, holder_character_sheet=sheet)
take_pledge(sheet, p2)
sacrifice_items(sheet, [item])
print("IVO", ivo.pk, entry.pk)

record = ScheduledTaskRecord.objects.get(task_key="worship.held_answers")
record.last_run_at = timezone.now() - timedelta(hours=25)
record.save(update_fields=["last_run_at"])
print("TASK last_run_at", record.last_run_at)
