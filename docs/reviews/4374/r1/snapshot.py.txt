"""Read-only snapshot: held answers, visions, the pool, and Ivo's knowledge."""
from evennia.objects.models import ObjectDB
from world.codex.models import CharacterCodexKnowledge
from world.clues.models import CharacterClue
from world.game_clock.models import ScheduledTaskRecord
from world.worship.models import PendingAnswer, Vision, WorshippedBeing

print("SNAP pool", WorshippedBeing.objects.get(name="Ombrin").resonance_pool,
      "task last_run_at", ScheduledTaskRecord.objects.get(task_key="worship.held_answers").last_run_at)
for p in PendingAnswer.objects.select_related("character_sheet__character", "answer__outcome_tier").order_by("pk"):
    print("SNAP held", p.pk, p.character_sheet.character.db_key, p.answer.outcome_tier.name, p.created_at)
for v in Vision.objects.select_related("recipient__character", "answer__outcome_tier").order_by("pk"):
    print("SNAP vision", v.pk, v.recipient.character.db_key, v.answer.outcome_tier.name if v.answer_id else "-", "clue", v.clue_id, v.sent_at)
ivo = ObjectDB.objects.filter(db_key="Ivo Marsh").first()
if ivo is not None:
    e = ivo.sheet_data.roster_entry
    print("SNAP ivo codex", CharacterCodexKnowledge.objects.filter(roster_entry=e).count(), "clues", CharacterClue.objects.filter(roster_entry=e).count())
    for v in Vision.objects.filter(recipient=ivo.sheet_data):
        print("SNAP ivo vision body", repr(v.body))
        print("SNAP ivo message", repr(v.message.body))
