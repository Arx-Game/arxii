"""Scratch: the two cron callables, run with the server stopped. First the held-answer tick
while the pool is still short, then with the pool able to pay; then the lapse."""
from datetime import timedelta

from django.utils import timezone
from evennia.objects.models import ObjectDB

from world.clues.models import CharacterClue
from world.codex.models import CharacterCodexKnowledge
from world.conditions.models import ConditionInstance
from world.missions.models import MissionInstance
from world.projects.services import scan_active_projects
from world.worship.answer_services import deliver_held_answers
from world.worship.models import DevotionStanding, PendingAnswer, PledgeDetails, Vision, WorshippedBeing

being = WorshippedBeing.objects.get(name="Ombrin")
mira = ObjectDB.objects.get(db_key="Mira Holt").sheet_data
def mira_state(label):
    print(label, "pool", WorshippedBeing.objects.get(pk=being.pk).resonance_pool,
          "held", PendingAnswer.objects.count(),
          "mira visions", Vision.objects.filter(recipient=mira).count(),
          "clues", CharacterClue.objects.filter(roster_entry=mira.roster_entry).count(),
          "codex", CharacterCodexKnowledge.objects.filter(roster_entry=mira.roster_entry).count(),
          "missions", MissionInstance.objects.filter(participants__character_id=mira.character.pk).count(),
          "conditions", ConditionInstance.objects.filter(target=mira.character).count())
mira_state("BEFORE")
print("TICK short pool delivered", deliver_held_answers())
mira_state("AFTER SHORT TICK")
being.resonance_pool = 100
being.save(update_fields=["resonance_pool"])
print("TICK paid pool delivered", deliver_held_answers())
mira_state("AFTER PAID TICK")
print("VISION", repr(Vision.objects.filter(recipient=mira).first().body))

wren = ObjectDB.objects.get(db_key="Wren Ashby").sheet_data
d = PledgeDetails.objects.get(character_sheet=wren)
fav = DevotionStanding.objects.get(character_sheet=wren, being=being).favor
print("WREN before", d.project.status, d.project.current_progress, "of", d.project.threshold_target, "favor", fav)
d.project.time_limit = timezone.now() - timedelta(minutes=1)
d.project.save(update_fields=["time_limit"])
print("LIFECYCLE TICK", scan_active_projects())
d.project.refresh_from_db()
print("WREN after", d.project.status, "favor", DevotionStanding.objects.get(character_sheet=wren, being=being).favor,
      "visions", Vision.objects.filter(recipient=wren).count(), "held", PendingAnswer.objects.filter(character_sheet=wren).count())
