from datetime import timedelta

from django.utils import timezone
from evennia.objects.models import ObjectDB

from world.magic.constants import GainSource
from world.magic.services.resonance import grant_resonance
from world.projects.services import scan_active_projects
from world.worship.constants import BeingResonanceTier
from world.worship.models import BeingResonance, DevotionStanding, PledgeCredit, PledgeDetails

sheet = lambda n: ObjectDB.objects.get(db_key=n).sheet_data
fav = BeingResonance.objects.get(tier=BeingResonanceTier.FAVORED).resonance
assoc = BeingResonance.objects.get(tier=BeingResonanceTier.ASSOCIATED).resonance
for name, res, amt in (
    ("Ysmay Dunhollow", fav, 3),
    ("Ysmay Dunhollow", assoc, 2),
    ("Orla Cobbe", fav, 8),
    ("Orla Cobbe", fav, 3),
):
    grant_resonance(sheet(name), res, amt, source=GainSource.STAFF_GRANT)
    d = PledgeDetails.objects.get(character_sheet=sheet(name))
    d.project.refresh_from_db()
    print("DEED", name, res.name, amt, "-> progress", d.project.current_progress)
for name in ("Tobiah Fenwright", "Hobb Pennock"):
    d = PledgeDetails.objects.get(character_sheet=sheet(name))
    d.project.time_limit = timezone.now() - timedelta(minutes=1)
    d.project.save(update_fields=["time_limit"])
print("TICK resolved", scan_active_projects())
for d in PledgeDetails.objects.select_related("project", "character_sheet").order_by("pk"):
    d.project.refresh_from_db()
    st = DevotionStanding.objects.get(character_sheet=d.character_sheet, being=d.pledge.being)
    print(
        "STATE",
        d.character_sheet.character.db_key,
        d.project.status,
        d.project.current_progress,
        "favor",
        st.favor,
    )
for c in PledgeCredit.objects.order_by("pk"):
    print("CREDIT", c.pledge_details.character_sheet.character.db_key, c.source, c.base, c.credited)
