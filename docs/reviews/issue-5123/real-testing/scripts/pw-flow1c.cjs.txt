const { open, login, EV } = require('./lib.cjs');
const fs = require('fs');
async function api(ctx, page, method, url, data) {
  const csrf = (await ctx.cookies()).find((c) => c.name === 'csrftoken')?.value;
  const H = { 'X-CSRFToken': csrf, 'Content-Type': 'application/json', Referer: 'http://localhost:4001/' };
  const r = await page.request.fetch(url, { method, headers: H, data: data ?? {} });
  return [r.status(), await r.text()];
}
(async () => {
  const out = []; const log = (s) => { console.log(s); out.push(s); };
  // The player submits her own draft, as the Review stage's Submit does.
  let { browser, page, ctx } = await open();
  await login(page, 'e2e_test_account', 'TestPass123!');
  let [s, t] = await api(ctx, page, 'POST', '/api/character-creation/drafts/1/submit/', { submission_notes: 'real-testing 5123' });
  log('player submit ' + s + ' ' + t.slice(0, 200));
  await browser.close();
  ({ browser, page, ctx } = await open());
  await login(page, 'rt_staff');
  [s, t] = await api(ctx, page, 'GET', '/api/character-creation/applications/');
  log('staff applications ' + s + ' ' + t.slice(0, 300));
  const apps = JSON.parse(t); const list = apps.results ?? apps;
  const id = list[0].id;
  [s, t] = await api(ctx, page, 'POST', `/api/character-creation/applications/${id}/claim/`);
  log('claim ' + s + ' ' + t.slice(0, 200));
  [s, t] = await api(ctx, page, 'POST', `/api/character-creation/applications/${id}/approve/`, { comment: 'ok' });
  log('approve ' + s + ' ' + t.slice(0, 300));
  fs.appendFileSync(`${EV}/flow1-web-transcript.txt`, '\n-- finalize --\n' + out.join('\n') + '\n');
  await browser.close();
})();
