const { open, login, shot, EV } = require('./lib.cjs');
const fs = require('fs');
const [user, entry, tag] = process.argv.slice(2);
(async () => {
  const out = []; const log = (s) => { console.log(s); out.push(s); };
  const { browser, page } = await open();
  await login(page, user);
  const aura = async (t) => { const a = await (await page.request.get('/api/magic/character-auras/')).json(); const r = (a.results ?? a)[0]; log(`${t}: unsettled=${r.unsettled} line="${r.unsettled_line}" mire="${r.mire_share_words}" dominant=${r.dominant_affinity} shares=${JSON.stringify(r.shares.map(s => [s.affinity_name, s.percent, s.settled_percent]))}`); };
  await aura('BEFORE');
  const pend = await (await page.request.get('/api/magic/drift-correction/pending/')).json();
  log('pending offers: ' + pend.count);
  await page.goto(`/characters/${entry}`);
  await page.waitForTimeout(2500);
  await page.getByText(/^magic$/i).first().click();
  await page.waitForTimeout(2500);
  await shot(page, `${tag}-01-sheet-unsettled`);
  await page.getByRole('button', { name: /settle your aura/i }).click();
  await page.waitForTimeout(2000);
  const dlg = page.getByRole('dialog');
  log('DIALOG: ' + (await dlg.innerText()).replace(/\n+/g, ' | '));
  await shot(page, `${tag}-02-dialog-accept-only`, { full: false });
  await dlg.getByRole('button', { name: /^accept$/i }).click();
  await page.waitForTimeout(3000);
  log('DIALOG AFTER ACCEPT: ' + ((await dlg.count()) ? (await dlg.first().innerText()).replace(/\n+/g, ' | ') : '(closed)'));
  await shot(page, `${tag}-03-after-accept`, { full: false });
  await aura('AFTER');
  await page.reload(); await page.waitForTimeout(2500);
  await page.getByText(/^magic$/i).first().click(); await page.waitForTimeout(2500);
  await shot(page, `${tag}-04-sheet-after-accept`);
  const t = await page.locator('body').innerText(); const i = t.indexOf('Aura');
  log('SHEET AURA TEXT: ' + t.slice(i, i + 200).replace(/\n+/g, ' | '));
  fs.writeFileSync(`${EV}/${tag}-web-transcript.txt`, out.join('\n') + '\n');
  await browser.close();
})();
