const { open, login, EV } = require('./lib.cjs');
const fs = require('fs');
(async () => {
  const { browser, page, ctx } = await open();
  await login(page, 'rt_ysmay');
  const csrf = (await ctx.cookies()).find((c) => c.name === 'csrftoken')?.value;
  const r = await page.request.post('/api/magic/thread-pull-preview/', {
    headers: { 'X-CSRFToken': csrf, 'Content-Type': 'application/json', Referer: 'http://localhost:4001/' },
    data: { character_sheet_id: 5, tier: 1, thread_ids: [1] },
  });
  console.log(r.status());
  const j = await r.json();
  console.log(JSON.stringify(j, null, 1).slice(0, 6000));
  fs.writeFileSync(`${EV}/flow2-pull-preview-api.json`, JSON.stringify(j, null, 1));
  await browser.close();
})();
