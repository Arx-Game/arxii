# Throwaway-database seed for the #5123 real-testing pass (rt_5123 only).
# Builds staff-authored rows and players with the test factories, in ONE transaction.
from decimal import Decimal

from django.db import transaction
from evennia.accounts.models import AccountDB

PW = "RtPass5123!"

with transaction.atomic():
    from actions.factories import ActionTemplateFactory
    from evennia_extensions.factories import CharacterFactory, RoomProfileFactory
    from evennia_extensions.models import PlayerData
    from integration_tests.live_seeds import _played_character
    from world.action_points.models import ActionPointConfig, ActionPointPool
    from world.character_creation.constants import Stage
    from world.character_creation.tests.finalization_fixtures import FinalizationTestMixin
    from world.character_sheets.factories import CharacterSheetFactory
    from world.checks.factories import CheckTypeFactory, CheckTypeTraitFactory
    from world.classes.factories import CharacterClassFactory
    from world.classes.services import set_primary_class_level
    from world.combat.constants import ActionCategory  # noqa
    from world.conditions.constants import DurationType
    from world.conditions.factories import (
        ConditionCategoryFactory,
        ConditionCheckModifierFactory,
        ConditionTemplateFactory,
    )
    from world.items.factories import ItemInstanceFactory, ItemTemplateFactory
    from world.items.services.materialize import materialize_item_game_object
    from world.magic.constants import EffectKind, GainSource, TargetKind
    from world.magic.factories import (
        BinaryEffectTypeFactory,
        CharacterAnimaFactory,
        CharacterAuraFactory,
        CharacterResonanceFactory,
        CharacterTechniqueFactory,
        DramaticMomentTypeFactory,
        DriftCorrectionRiteFactory,
        GiftFactory,
        ImbuingRitualFactory,
        ResonanceFactory,
        ResonanceTierFactory,
        TechniqueFactory,
        ThreadFactory,
        ThreadPullCostFactory,
        ThreadPullEffectFactory,
        ThreadRetuneRiteFactory,
        ensure_conversion_pairs,
        set_aura_split,
        _make_resist_corruption_check_type,
    )
    from world.magic.models import ResonanceConversion
    from world.magic.services.drift import get_drift_config, get_drift_correction_config
    from world.magic.services.resonance import grant_resonance
    from world.magic.specialization.services import provision_latent_gift_thread
    from world.checks.models import CheckType  # noqa
    from world.roster.services.selection import set_selected_entry
    from world.scenes.factories import SceneFactory, SceneParticipationFactory
    from world.scenes.interaction_services import invalidate_active_scene_cache
    from world.seeds.test_account import seed_test_account
    from world.traits.constants import CANONICAL_OUTCOME_LEVELS
    from world.traits.factories import CheckSystemSetupFactory, StatTraitFactory
    from world.traits.services import ensure_canonical_check_outcome
    from world.magic.models import DriftCorrectionOutcomeShare
    from world.vitals.models import CharacterVitals
    from world.scenes.tests.cast_test_helpers import grant_technique
    from evennia.utils import create

    celestial, primal, abyssal = ensure_conversion_pairs()
    CheckSystemSetupFactory.create()
    _make_resist_corruption_check_type()
    ActionPointConfig.objects.update_or_create(
        name="Default", defaults={"is_active": True, "weekly_regen": 80}
    )

    # ---- resonances (staff catalog) ------------------------------------------------
    fury = ResonanceFactory(name="Fury", affinity=primal, description="Rage that drives.")
    wonder = ResonanceFactory(name="Wonder", affinity=primal, description="Awe that opens doors.")
    domination = ResonanceFactory(
        name="Domination", affinity=abyssal, description="The need to bend others."
    )
    dread = ResonanceFactory(name="Dread", affinity=abyssal, description="Fear that spreads.")
    mercy = ResonanceFactory(name="Mercy", affinity=celestial, description="Sparing the fallen.")
    for source, target_affinity, target in (
        (domination, primal, fury),
        (dread, primal, fury),
        (fury, abyssal, domination),
    ):
        ResonanceConversion.objects.get_or_create(
            source_resonance=source, target_affinity=target_affinity, target_resonance=target
        )

    # ---- drift config / focus check / atone rite (staff-authored) ------------------
    cast_type = CheckTypeFactory()
    CheckTypeTraitFactory(check_type=cast_type, trait=StatTraitFactory(), weight=1)
    focus_type = CheckTypeFactory(name="Keeping Focus")
    CheckTypeTraitFactory(check_type=focus_type, trait=StatTraitFactory(), weight=1)
    conflict = ConditionTemplateFactory(
        name="Resonance Conflict",
        category=ConditionCategoryFactory(self_shakeable=True),
        default_duration_type=DurationType.SCENE,
    )
    ConditionCheckModifierFactory(condition=conflict, check_type=cast_type, modifier_value=-10)
    cfg = get_drift_config()
    cfg.focus_check_type = focus_type
    cfg.conflict_condition = conflict
    cfg.unsettled_tolerance_percent = 5
    cfg.save()
    dcc = get_drift_correction_config()
    dcc.rite = DriftCorrectionRiteFactory()
    dcc.check_type = CheckTypeFactory()
    dcc.check_difficulty = 0
    dcc.cap_percent = 50
    dcc.save()
    for name in CANONICAL_OUTCOME_LEVELS:
        DriftCorrectionOutcomeShare.objects.update_or_create(
            outcome_tier=ensure_canonical_check_outcome(name),
            defaults={"converted_share": Decimal("1.00")},
        )

    # ---- rites, pull costs and perks ------------------------------------------------
    imbue_rite = ImbuingRitualFactory()
    retune_rite = ThreadRetuneRiteFactory()
    ThreadPullCostFactory(tier=1, resonance_cost=3, anima_per_thread=1)
    # Gift-thread perks: one Fury perk, one Domination perk, two rungs each. Wonder none.
    for res, low, high, text_low, text_high in (
        (fury, 2, 5, "A surge of temper sharpens the working.", "Fury floods the working."),
        (domination, 3, 7, "Your will presses on the working.", "Your will bends the working."),
    ):
        for min_level, amount, text in ((0, low, text_low), (10, high, text_high)):
            ThreadPullEffectFactory(
                target_kind=TargetKind.GIFT, resonance=res, tier=1, min_thread_level=min_level,
                effect_kind=EffectKind.FLAT_BONUS, flat_bonus_amount=amount,
                narrative_snippet=text,
            )
    # Technique-thread perks (for the retune flow): Domination and Dread each have one.
    for res, amount in ((domination, 4), (dread, 6)):
        ThreadPullEffectFactory(
            target_kind=TargetKind.TECHNIQUE, resonance=res, tier=1, min_thread_level=0,
            effect_kind=EffectKind.FLAT_BONUS, flat_bonus_amount=amount,
            narrative_snippet=f"{res.name} colours the technique.",
        )

    # ---- the Lantern: a room with a live scene --------------------------------------
    lantern = create.create_object("typeclasses.rooms.Room", key="The Lantern")
    quiet = create.create_object("typeclasses.rooms.Room", key="The Quiet Room")
    scene = SceneFactory(
        location=RoomProfileFactory(objectdb=lantern), is_active=True, name="Lantern Night"
    )
    invalidate_active_scene_cache(lantern)

    gift = GiftFactory(name="Kindling")
    gift.resonances.add(fury, domination, wonder)
    technique = TechniqueFactory(
        name="Lantern Ward", gift=gift, anima_cost=1, effect_type=BinaryEffectTypeFactory(),
        damage_profile=False, action_template=ActionTemplateFactory(check_type=cast_type),
    )

    # a fury-tied offering and a Dread-tied offering for the rites
    def carry(sheet, resonance, name):
        template = ItemTemplateFactory(
            name=name, tied_resonance=resonance, resonance_tier=ResonanceTierFactory()
        )
        item = ItemInstanceFactory(template=template, holder_character_sheet=sheet)
        materialize_item_game_object(item, sheet)
        return item.pk

    def player(username, name, room, *, scene_member=False, level=3):
        seed_test_account(username=username, email=f"{username}@example.com", password=PW)
        account = AccountDB.objects.get(username=username)
        tenure = _played_character(account, name, room)
        character = tenure.roster_entry.character_sheet.character
        sheet = character.sheet_data
        set_selected_entry(tenure.player_data, tenure.roster_entry)
        set_primary_class_level(character, CharacterClassFactory(), level)
        CharacterAnimaFactory(character=sheet, current=40, maximum=40)
        CharacterVitals.objects.create(
            character_sheet=sheet, health=50, max_health=50, base_max_health=50
        )
        pool = ActionPointPool.get_or_create_for_character(character)
        pool.current = 500
        pool.maximum = 500
        pool.save()
        if scene_member:
            SceneParticipationFactory(scene=scene, account=account)
        return account, character, sheet

    def hold(sheet, resonance, balance, lifetime=None):
        return CharacterResonanceFactory(
            character_sheet=sheet, resonance=resonance, balance=balance,
            lifetime_earned=balance if lifetime is None else lifetime,
        )

    # ---- Ysmay: the loom (flows 2, 6, 7) --------------------------------------------
    acc, ysmay, ys = player("rt_ysmay", "Ysmay", lantern, scene_member=True)
    set_aura_split(CharacterAuraFactory(character=ys), celestial="0", primal="70", abyssal="30")
    hold(ys, fury, 60)
    hold(ys, wonder, 10)
    hold(ys, domination, 30)
    hold(ys, dread, 10)
    grant_technique(ys.primary_persona, technique)
    gift_thread = provision_latent_gift_thread(ys, gift, resonances=[fury, domination, wonder])
    ThreadFactory(
        owner=ys, resonance=domination, target_kind=TargetKind.TECHNIQUE, target_trait=None,
        target_technique=technique, level=20, developed_points=40, name="Loom Strand",
    )
    retune_offering = carry(ys, dread, "Dread Shard")

    # ---- Veyra: becomes unsettled, then atones (flows 3, 4) ---------------------------
    acc, veyra, vs = player("rt_veyra", "Veyra", lantern, scene_member=True)
    set_aura_split(CharacterAuraFactory(character=vs), celestial="0", primal="90", abyssal="10")
    hold(vs, fury, 200)
    hold(vs, domination, 10)
    grant_technique(vs.primary_persona, technique)
    veyra_offering = carry(vs, fury, "Fury Ember")
    DramaticMomentTypeFactory(
        label="Fury Flare", resonance=fury, per_scene_cap=3, resonance_amount=20
    )

    # ---- Hobb: already unsettled, outside any scene (flow 4, Accept only) -------------
    acc, hobb, hs = player("rt_hobb", "Hobb", quiet)
    set_aura_split(CharacterAuraFactory(character=hs), celestial="0", primal="90", abyssal="10")
    hold(hs, fury, 200)
    hold(hs, domination, 50)

    # ---- Marrow: staff who tags moments (a deed) --------------------------------------
    acc, marrow, ms = player("rt_marrow", "Marrow", lantern, scene_member=True, level=1)
    acc.is_staff = True
    acc.save(update_fields=["is_staff"])
    acc.permissions.add("Admin")
    set_aura_split(CharacterAuraFactory(character=ms), celestial="100", primal="0", abyssal="0")

    # ---- Sorrel: a Sineater taking Mire (flow 5) --------------------------------------
    acc, sorrel, ss = player("rt_sorrel", "Sorrel", quiet)
    set_aura_split(CharacterAuraFactory(character=ss), celestial="100", primal="0", abyssal="0")
    dawnlight = ResonanceFactory(name="Dawnlight", affinity=celestial)
    hold(ss, dawnlight, 700)
    grant_resonance(ss, domination, 200, source=GainSource.SINEATING, from_mire=True)

    # ---- the creation draft: e2e_test_account's Gift stage -----------------------------
    class _CG(FinalizationTestMixin):
        pass

    cg = _CG()
    cg.account = AccountDB.objects.get(username="e2e_test_account")
    _CG._setup_finalization_base(cg, prefix="Rt", height_min=700, height_max=800)
    draft = cg._create_base_draft(first_name="Brannoch")
    # The base setup made one random-named resonance of its own; drop it from the catalog
    # and clear the draft's picks so the player makes them.
    stray = cg.resonance
    draft.draft_data["gift_resonance_weights"] = []
    draft.current_stage = Stage.GIFT
    draft.save()
    gift.resonances.add(mercy)
    cg.gift.resonances.add(fury, wonder, domination, mercy)
    from world.magic.models import Resonance
    Resonance.objects.filter(pk=stray.pk).delete()

    print("SEEDED", {
        "scene": scene.pk, "gift_thread": gift_thread.pk, "draft": draft.pk,
        "retune_offering": retune_offering, "veyra_offering": veyra_offering,
        "ysmay": ysmay.pk, "veyra": veyra.pk, "hobb": hobb.pk, "marrow": marrow.pk,
        "sorrel": sorrel.pk, "technique": technique.pk,
    })
