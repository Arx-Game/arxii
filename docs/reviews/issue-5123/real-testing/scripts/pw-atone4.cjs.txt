const { open, login, shot, EV } = require('./lib.cjs');
const fs = require('fs');
(async () => {
  const out = []; const log = (s) => { console.log(s); out.push(s); };
  const { browser, page } = await open();
  await login(page, 'rt_veyra');
  const aura = async (tag) => { const a = await (await page.request.get('/api/magic/character-auras/')).json(); const r = (a.results ?? a)[0]; log(`${tag}: unsettled=${r.unsettled} line="${r.unsettled_line}" shares=${JSON.stringify(r.shares.map(s => [s.affinity_name, s.percent, s.settled_percent]))}`); };
  const res = async (tag) => { const r = await (await page.request.get('/api/magic/character-resonances/')).json(); log(`${tag}: ` + JSON.stringify((r.results ?? r).map(x => [x.resonance_name, 'life', x.lifetime_earned, 'bal', x.balance]))); };
  await aura('BEFORE atone'); await res('BEFORE atone');
  await page.goto('/scenes/1');
  await page.waitForTimeout(5000);
  const dlg = page.getByRole('dialog');
  await dlg.getByText('Fury Ember').click();
  await page.waitForTimeout(500);
  await shot(page, 'flow4-04-offering-picked', { full: false });
  await dlg.getByRole('button', { name: /begin the rite/i }).click();
  await page.waitForTimeout(4000);
  log('DIALOG AFTER RITE: ' + ((await dlg.count()) ? (await dlg.first().innerText()).replace(/\n+/g, ' | ') : '(closed)'));
  await shot(page, 'flow4-05-after-rite', { full: false });
  await aura('AFTER atone'); await res('AFTER atone');
  fs.appendFileSync(`${EV}/flow4-web-atone-transcript.txt`, out.join('\n') + '\n');
  await browser.close();
})();
