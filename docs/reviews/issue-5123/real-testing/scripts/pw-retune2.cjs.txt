const { open, login, shot, EV } = require('./lib.cjs');
const fs = require('fs');
(async () => {
  const out = []; const log = (s) => { console.log(s); out.push(s); };
  const { browser, page, ctx } = await open();
  await login(page, 'rt_ysmay');
  const csrf = (await ctx.cookies()).find((c) => c.name === 'csrftoken')?.value;
  const preview = async (tag) => {
    const r = await page.request.post('/api/magic/thread-pull-preview/', { headers: { 'X-CSRFToken': csrf, 'Content-Type': 'application/json', Referer: 'http://localhost:4001/' }, data: { character_sheet_id: 5, tier: 1, thread_ids: [2] } });
    const j = await r.json();
    log(`${tag} pull preview (thread 2): ` + JSON.stringify(j.per_thread?.[0]?.options.map(o => [o.resonance_name, 'level_read', o.level_read, 'perks', o.effects.map(e => e.scaled_value)])));
  };
  const th = async (tag) => { const t = await (await page.request.get('/api/magic/threads/2/')).json(); const all = await (await page.request.get('/api/magic/threads/')).json(); log(`${tag} thread: level=${t.level} developed=${t.developed_points} leads=${t.resonance_name} rows=${JSON.stringify(t.resonances.map(r => [r.resonance_name, r.amount]))}; thread count for Ysmay=${(all.results ?? all).length}`); };
  await th('BEFORE'); await preview('BEFORE');
  await page.goto('/threads/2');
  await page.waitForTimeout(2500);
  await page.getByRole('button', { name: /change resonance/i }).click();
  await page.waitForTimeout(2000);
  const dlg = page.getByTestId('thread-resonance-dialog');
  for (let i = 0; i < 35; i++) await dlg.getByTestId('resonance-amount-down').click();
  log('points to move shown: ' + await dlg.getByTestId('resonance-amount-value').innerText());
  await dlg.getByText(/^Dread/).first().click();
  await dlg.getByText(/Dread Shard/).first().click();
  await page.waitForTimeout(800);
  log('PROJECTED: ' + await dlg.getByTestId('resonance-projected').innerText());
  await shot(page, 'flow6-03-retune-five-to-dread', { full: false });
  await dlg.getByTestId('resonance-retune').click();
  await page.waitForTimeout(3500);
  log('DIALOG AFTER: ' + ((await page.getByRole('dialog').count()) ? (await page.getByRole('dialog').first().innerText()).replace(/\n+/g, ' | ') : '(closed)'));
  await shot(page, 'flow6-04-after-retune', { full: false });
  await th('AFTER'); await preview('AFTER');
  await page.goto('/threads/2'); await page.waitForTimeout(2500);
  await shot(page, 'flow6-05-thread-page-after-retune');
  fs.writeFileSync(`${EV}/flow6-web-transcript.txt`, out.join('\n') + '\n');
  await browser.close();
})();
