const { open, login, shot, EV } = require('./lib.cjs');
const fs = require('fs');
(async () => {
  const out = []; const log = (s) => { console.log(s); out.push(s); };
  const { browser, page } = await open();
  await login(page, 'rt_veyra');
  await page.goto('/scenes/1');
  await page.waitForTimeout(6000);
  await shot(page, 'flow4-03-scene-page-offer', { full: false });
  const dlg = page.getByRole('dialog');
  log('dialog count: ' + await dlg.count());
  if (await dlg.count()) log('DIALOG: ' + (await dlg.first().innerText()).replace(/\n+/g, ' | '));
  else log('BODY: ' + (await page.locator('body').innerText()).slice(0, 800).replace(/\n+/g, ' | '));
  fs.appendFileSync(`${EV}/flow4-web-atone-transcript.txt`, out.join('\n') + '\n');
  await browser.close();
})();
