const { open, login, shot, EV } = require('./lib.cjs');
const fs = require('fs');
(async () => {
  const out = []; const log = (s) => { console.log(s); out.push(s); };
  const { browser, page } = await open();
  await login(page, 'rt_veyra');
  await page.goto('/characters/2');
  await page.waitForTimeout(2500);
  const enter = page.getByText(/enter the world|return to the world/i).first();
  log('world button: ' + (await enter.innerText()));
  await enter.click();
  await page.waitForTimeout(5000);
  await page.goto('/characters/2');
  await page.waitForTimeout(2500);
  await page.getByText(/^magic$/i).first().click();
  await page.waitForTimeout(2500);
  const pend = await (await page.request.get('/api/magic/drift-correction/pending/')).text();
  log('PENDING (api) = ' + pend.slice(0, 1800));
  await shot(page, 'flow4-01-veyra-sheet-with-open-offer');
  await page.getByRole('button', { name: /settle your aura/i }).click();
  await page.waitForTimeout(2500);
  await shot(page, 'flow4-02-settle-your-aura-dialog', { full: false });
  log('DIALOG: ' + (await page.getByRole('dialog').innerText()).replace(/\n+/g, ' | '));
  fs.writeFileSync(`${EV}/flow4-web-atone-transcript.txt`, out.join('\n') + '\n');
  await browser.close();
})();
