const { open, login, shot, EV } = require('./lib.cjs');
const fs = require('fs');
const [user, n] = [process.argv[2], parseInt(process.argv[3] || '3')];
(async () => {
  const out = []; const log = (s) => { console.log(s); out.push(s); };
  const { browser, page } = await open();
  await login(page, user);
  for (let i = 1; i <= n; i++) {
    await page.goto('/scenes/1');
    await page.waitForTimeout(2000);
    await page.locator('button:has(svg.lucide-swords)').click();
    await page.getByRole('button', { name: /Cast/ }).click();
    await page.waitForTimeout(800);
    await page.locator('#cast-section').getByRole('button', { name: /Lantern Ward/ }).first().click();
    await page.waitForTimeout(1500);
    await page.getByRole('button', { name: /^Cast Lantern Ward/ }).click();
    await page.waitForTimeout(3500);
    const led = page.getByTestId('cast-ledger-result');
    log(`${user} cast #${i}: ` + ((await led.count()) ? (await led.innerText()).replace(/\n+/g, ' | ') : '(none)'));
    const body = (await page.locator('body').innerText());
    const f = body.match(/[^\n]*focus[^\n]*/gi);
    log('  focus lines in page: ' + JSON.stringify(f));
    if (i === 1) await shot(page, `flow3-03-${user}-first-cast`);
  }
  fs.appendFileSync(`${EV}/flow3-casts-web-transcript.txt`, out.join('\n') + '\n');
  await browser.close();
})();
