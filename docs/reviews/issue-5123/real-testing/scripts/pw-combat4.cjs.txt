const { open, login, shot, EV } = require('./lib.cjs');
const fs = require('fs');
(async () => {
  const out = []; const log = (s) => { console.log(s); out.push(s); };
  const { browser, page } = await open(1440, 1000);
  await login(page, 'rt_ysmay');
  await page.goto('/characters/1'); await page.waitForTimeout(2500);
  await page.getByText(/enter the world|return to the world/i).first().click();
  await page.waitForTimeout(6000);
  await page.getByTestId('open-pull-dialog-btn').click();
  await page.waitForTimeout(2500);
  await page.getByTestId('tier-btn-1').click();
  await page.getByTestId('thread-checkbox-1').click();
  await page.waitForTimeout(3000);
  const dlg = page.getByTestId('thread-pull-dialog');
  log('DIALOG TEXT: ' + (await dlg.innerText()).replace(/\n+/g, ' | '));
  const css = async (sel, label) => { const c = await page.locator(sel).first().evaluate(e => { const s = getComputedStyle(e); return { bg: s.backgroundColor, color: s.color, border: s.borderColor }; }); log(`COLOUR ${label}: ${JSON.stringify(c)}`); };
  await css('[data-testid=tier-btn-1]', 'selected tier button');
  await css('[data-testid=tier-btn-2]', 'unselected tier button');
  await css('[data-testid=commit-pull-btn]', 'Select Pull confirm button');
  const vars = await page.evaluate(() => { const s = getComputedStyle(document.documentElement); return { primary: s.getPropertyValue('--primary'), realm: document.documentElement.getAttribute('data-realm') || document.body.getAttribute('data-realm') }; });
  log('ROOT VARS ' + JSON.stringify(vars));
  await shot(page, 'flow2-07-combat-pull-threads-dialog-tier1-gift-thread', { full: false });
  // payer choice: click the Fury option if present and capture
  const fury = dlg.getByRole('radio', { name: /^Fury/ });
  if (await fury.count()) { await fury.first().click(); await page.waitForTimeout(800); await shot(page, 'flow2-08-combat-dialog-fury-pays', { full: false }); log('after Fury picked: ' + (await dlg.innerText()).replace(/\n+/g, ' | ')); }
  fs.writeFileSync(`${EV}/flow2-combat-dialog-transcript.txt`, out.join('\n') + '\n');
  await browser.close();
})();
