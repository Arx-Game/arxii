async function mockCg(page: Page, weights: { resonance_id: number; weight: number }[]) {
  const draft = createMockDraft({
    id: 7,
    current_stage: 6,
    selected_area: mockStartingArea,
    selected_beginnings: mockBeginnings,
    selected_path: mockPath,
    selected_tradition: mockTradition,
    stage_completion: { 1: true, 2: true, 3: true, 4: true, 5: true, 6: false, 7: false, 8: false, 9: false, 10: false, 11: false } as never,
    draft_data: { selected_gift_id: 1, selected_technique_ids: [10], gift_resonance_weights: weights },
  });
  const unknown: string[] = [];
  await page.route((url) => url.pathname.startsWith('/api/'), async (route: Route) => {
    const u = new URL(route.request().url());
    const p = u.pathname;
    const method = route.request().method();
    const j = (json: unknown) => route.fulfill({ json });
    if (p === '/api/user/') return j(account);
    if (p === '/api/roster/entries/mine/') return j([]);
    if (p === '/api/backgrounds/') return j([]);
    if (p === '/api/character-creation/can-create/') return j({ can_create: true, reason: '' });
    if (p === '/api/character-creation/explanations/') return j(mockCGExplanations);
    if (p === '/api/character-creation/drafts/' && method === 'GET') return j([draft]);
    if (p === '/api/character-creation/drafts/7/' && method === 'PATCH') {
      const body = route.request().postDataJSON() as { draft_data?: Record<string, unknown> };
      if (body.draft_data) Object.assign(draft.draft_data, body.draft_data);
      return j(draft);
    }
    if (p === '/api/character-creation/drafts/7/') return j(draft);
    if (p === '/api/magic/resonances/') return j(RES);
    if (p === '/api/character-creation/cg-budgets/') return j({ starting_points: 100 });
    if (p === '/api/character-creation/drafts/7/cg-points/')
      return j({ spent: 0, remaining: 100, budget: 100, lines: [] });
    if (p === '/api/character-creation/traditions/') return j([mockTradition]);
    if (p === '/api/character-creation/gifts/') return j(mockCGGiftOptions);
    if (p === '/api/character-creation/combat-styles/') return j([]);
    if (p === '/api/character-creation/technique-options/') return j(mockCGTechniqueOptions);
    if (p === '/api/roster/mail/unread-count/') return j({ count: 0 });
    if (p === '/api/character-creation/glimpse-tags/') return j([]);
    if (p === '/api/character-creation/drafts/7/offers/') return j({ offers: [], terms: [] });
    if (p.endsWith('/perspectives/')) return j([]);
    if (p.includes('stat-definitions') || p.includes('with_specializations')) return j([]);
    unknown.push(`${method} ${p}`);
    if (method === 'GET') return j([]);
    return route.fulfill({ status: 404, json: { detail: 'Not provided by this fixture.' } });
  });
  return { draft, unknown };
}

test('screen 4: atone through the offer gate, offer spent, aura refetched', async ({ page }) => {
  const cur = { spent: false };
  const aura = (settled: boolean) => [{ id: 1, character: 1,
    shares: [['Celestial', 1, 0, 0], ['Primal', 2, settled ? 66 : 58, 66], ['Abyssal', 3, settled ? 34 : 42, 34]].map(([n, id, pct, st]) => ({
      affinity_id: id, affinity_name: n, percent: String(pct), settled_percent: String(st) })),
    dominant_affinity: 'Primal', dominant_affinity_display: 'Primal', unsettled: !settled,
    unsettled_line: settled ? '' : 'Your Domination has grown strong enough to make you unsettled.',
    unsettled_affinity_name: settled ? '' : 'Abyssal', mire_share_words: 'none', glimpse_state: 'NOT_STARTED', updated_at: '2026-11-14T00:00:00Z' }];
  const offer = { id: 5, scene_id: 9, character_id: 1, rite_ready: true, rite_not_ready_message: '',
    caps: [{ target_resonance: { id: 11, name: 'Fury' }, cap_remaining: 4, focus_bonus: 0,
      sources: [{ resonance_id: 13, resonance_name: 'Domination', balance: 4, multiplier: '1.00', conversion_type: 'FALL', max_amount: 4, projected_output: 4 }],
      qualifying_components: [{ item_id: 71, name: 'Ember censer', mode: 'burn' }] }] };
  const posts: string[] = [];
  await page.route((url) => url.pathname.startsWith('/api/'), async (route: Route) => {
    const p = new URL(route.request().url()).pathname;
    const method = route.request().method();
    const j = (json: unknown) => route.fulfill({ json });
    if (p === '/api/user/') return j(account);
    if (p === '/api/magic/drift-correction/pending/')
      return j({ count: cur.spent ? 0 : 1, next: null, previous: null, results: cur.spent ? [] : [offer] });
    if (p === '/api/magic/character-auras/') return j(aura(cur.spent));
    if (p === '/api/magic/drift-correction/convert/' && method === 'POST') {
      posts.push(route.request().postData() ?? '');
      cur.spent = true;
      return j({ source_resonance_id: 13, target_resonance_id: 11, attempted_amount: 4, converted_amount: 4, granted_amount: 4,
        outcome_name: 'Success', success_level: 1, conversion_type: 'FALL', focus_bonus: 0 });
    }
    if (method === 'GET') return j([]);
    return route.fulfill({ status: 404, json: { detail: 'Not provided by this fixture.' } });
  });
  const errors: string[] = [];
  page.on('pageerror', (e) => errors.push(e.message));
  await page.goto('/review5123.html?s=gate');
  await expect(page.getByTestId('settle-exit-atone')).toBeVisible();
  await expect(page.getByTestId('settle-exit-accept')).toBeVisible();
  await page.waitForTimeout(500);
  await page.screenshot({ path: shot('app-screen4-gate-open-offer.png') });
  await page.getByTestId('drift-component-71').click();
  await page.getByTestId('drift-perform-11').click();
  await expect(page.getByTestId('drift-result')).toBeVisible();
  await page.waitForTimeout(800);
  // The offer is spent and the character settled: neither exit remains, only the result.
  await expect(page.getByTestId('settle-exit-atone')).toHaveCount(0);
  await expect(page.getByTestId('settle-exit-accept')).toHaveCount(0);
  await page.screenshot({ path: shot('app-screen4-gate-after-atone.png') });
  console.log('CONVERT POSTS', posts.join(' | '), await page.getByTestId('drift-result').innerText());
  await page.getByTestId('settle-not-now').click();
  await page.waitForTimeout(400);
  await page.screenshot({ path: shot('app-screen4-gate-closed.png') });
  await expect(page.getByTestId('drift-conversion-gate-strip')).toHaveCount(0);
  console.log('ERRORS', JSON.stringify(errors));
});
