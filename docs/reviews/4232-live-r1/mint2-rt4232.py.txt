import os
import sys
from types import SimpleNamespace

os.environ.setdefault("DJANGO_SETTINGS_MODULE", "server.conf.settings")
import django

django.setup()
from django.conf import settings  # noqa: E402

if settings.DATABASES["default"]["NAME"] != "rt_4232":
    sys.exit("refusing: not rt_4232")
import evennia  # noqa: E402

evennia._init()
import evennia.accounts.accounts as accounts_mod  # noqa: E402

accounts_mod._SESSIONS = SimpleNamespace(sessions_from_account=lambda a: [])
from evennia.accounts.models import AccountDB  # noqa: E402
from evennia.objects.models import ObjectDB  # noqa: E402

from world.ceremonies.seeds import seed_ceremony_types  # noqa: E402
from world.roster.models import UnionKind  # noqa: E402
from world.roster.services.kinship import ensure_node_for_sheet  # noqa: E402
from world.roster.services.staff_characters import (  # noqa: E402
    _mint_character_working_set,
    mint_gm_character,
)
from world.seeds.test_account import seed_test_account  # noqa: E402
from world.worship.factories import WorshippedBeingFactory  # noqa: E402
from world.worship.models import WorshipDeclaration  # noqa: E402

setup_char = ObjectDB.objects.filter(pk=1).first()
if setup_char is not None and setup_char.db_typeclass_path.endswith("characters.Character"):
    setup_char.delete()
staff = AccountDB.objects.get(username="e2e_test_account")
rowan = mint_gm_character(staff, "Rowan Ashby")
seed_test_account(username="e2e_lisbet", email="lisbet@example.com", password="TestPass123!")
lisbet_account = AccountDB.objects.get(username="e2e_lisbet")
lisbet = _mint_character_working_set(
    lisbet_account, "Lisbet Crane", "typeclasses.characters.Character"
)
seed_test_account(username="e2e_orrin", email="orrin@example.com", password="TestPass123!")
orrin_account = AccountDB.objects.get(username="e2e_orrin")
orrin = _mint_character_working_set(
    orrin_account, "Father Orrin", "typeclasses.characters.Character"
)
being = WorshippedBeingFactory(name="The Lantern")
WorshipDeclaration.objects.create(character_sheet=orrin.character_sheet, public_being=being)
rowan_node = ensure_node_for_sheet(rowan.character_sheet)
lisbet_node = ensure_node_for_sheet(lisbet.character_sheet)
kind, _ = UnionKind.objects.get_or_create(name="Marriage", defaults={"confers_wedlock": True})
seed_ceremony_types()
print(
    "rowan",
    rowan.pk,
    "entry",
    rowan.character_sheet.roster_entry.pk,
    "node",
    rowan_node.pk,
    "| lisbet",
    lisbet.pk,
    "node",
    lisbet_node.pk,
    "| orrin",
    orrin.pk,
    "| being",
    being.pk,
    "kind",
    kind.pk,
    "| rooms",
    rowan.location,
    lisbet.location,
    orrin.location,
)
