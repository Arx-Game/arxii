import re
import socket
import sys
import time

cmds = sys.argv[1:]
s = socket.create_connection(("localhost", 4000))
s.settimeout(0.5)


def drain(seconds):
    end = time.time() + seconds
    out = b""
    while time.time() < end:
        try:
            d = s.recv(65536)
            if not d:
                break
            out += d
        except TimeoutError:
            pass
    return out


drain(1.5)
s.sendall(b"connect e2e_test_account TestPass123!\r\n")
drain(2.5)
for c in cmds:
    s.sendall(c.encode() + b"\r\n")
    raw = drain(2.5)
    text = re.sub(rb"\xff[\xfb-\xfe].", b"", raw)
    text = re.sub(rb"\x1b\[[0-9;]*m", b"", text)
    print("=== " + c)
    print(text.decode("latin1", "replace").strip())
s.sendall(b"quit\r\n")
drain(1)
s.close()
