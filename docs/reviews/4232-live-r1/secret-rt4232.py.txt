"""A secret marriage for Rowan (#4232 round 1, item 5). MODE=create|grant|death."""

import os
import sys

os.environ.setdefault("DJANGO_SETTINGS_MODULE", "server.conf.settings")
import django

django.setup()
from django.conf import settings

if settings.DATABASES["default"]["NAME"] != "rt_4232":
    sys.exit("refusing: not rt_4232")
import evennia

evennia._init()
from world.roster.models import Kinsperson, UnionKind
from world.roster.services.kinship import create_person, record_union
from world.secrets.constants import SecretLevel, SecretProvenance
from world.secrets.models import Secret
from world.secrets.services import author_secret, grant_secret_knowledge

rowan = Kinsperson.objects.get(pk=1)
mode = os.environ.get("MODE", "create")
if mode == "create":  # noqa: STRING_LITERAL - throwaway evidence script
    secret = author_secret(
        subject_sheet=rowan.sheet,
        provenance=SecretProvenance.GM_AUTHORED,
        level=SecretLevel.UNCOMMON_KNOWLEDGE,
        content="Rowan Ashby was wed in secret, years ago, to a veiled stranger.",
    )
    secret.subject_aware = False
    secret.save(update_fields=["subject_aware"])
    spouse = create_person(name="Veiled Spouse")
    union = record_union(
        kind=UnionKind.objects.get(name="Marriage"),
        members=[rowan, spouse],
        is_public_record=False,
        secret=secret,
    )
    print(
        "secret",
        secret.pk,
        "union",
        union.pk,
        "spouse",
        spouse.pk,
        "public",
        union.is_public_record,
    )
elif mode == "grant":  # noqa: STRING_LITERAL - throwaway evidence script
    secret = Secret.objects.get(subject_sheet=rowan.sheet)
    grant_secret_knowledge(roster_entry=rowan.sheet.roster_entry, secret=secret)
    print("granted secret", secret.pk, "to Rowan's own entry")
elif mode == "death":  # noqa: STRING_LITERAL - throwaway evidence script
    spouse = Kinsperson.objects.get(name="Veiled Spouse")
    spouse.believed_deceased = True
    spouse.save(update_fields=["believed_deceased"])
    print("Veiled Spouse believed dead; is_deceased =", spouse.is_deceased)
