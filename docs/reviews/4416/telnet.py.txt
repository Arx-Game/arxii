"""Real telnet sessions on :4000 for the #4416 pass. The GM (gm_4416 as Warden Ashe) stays
connected; the player (e2e_test_account) switches characters. Steps are JSON:
[["player"|"gm", "<command>" | "@ic <name>"], ...]. "{prompt}" and "{answer}" in a command
are filled from the GM's last output. Prints everything each session received."""
import json
import re
import socket
import sys
import time


def clean(buf):
    text = re.sub(rb"\xff[\xfb-\xfe].|\xff\xfa.*?\xff\xf0|\xff.", b"", buf, flags=re.S)
    return re.sub(r"\x1b\[[0-9;]*m", "", text.decode("utf-8", "replace"))


class Session:
    def __init__(self, label, user, char):
        self.label = label
        self.s = socket.create_connection(("localhost", 4000), timeout=5)
        self.read(2)
        self.send(f"connect {user} TestPass123!")
        print(f"[{label}] @ic {char}: " + self.send(f"@ic {char}").strip()[-200:])

    def read(self, wait=1.5):
        end = time.time() + wait
        buf = b""
        while time.time() < end:
            try:
                self.s.settimeout(0.3)
                chunk = self.s.recv(65536)
                if not chunk:
                    break
                buf += chunk
            except socket.timeout:
                pass
        return clean(buf)

    def send(self, cmd, wait=3.0):
        self.s.sendall((cmd + "\r\n").encode())
        return self.read(wait)


plan = json.loads(open(sys.argv[1]).read())
gm = Session("gm", "gm_4416", "Warden Ashe")
player = Session("player", "e2e_test_account", plan["first"])
found = {"prompt": "", "answer": ""}
for who, cmd in plan["steps"]:
    session = gm if who == "gm" else player
    cmd = cmd.format(**found)
    print(f"===== [{who}] > {cmd}")
    out = session.send(cmd, wait=4.0).strip()
    print(out)
    if who == "gm":
        ids = re.findall(r"GM prompt \[(\d+)\]|^\[(\d+)\] Being answer", out, flags=re.M)
        for a, b in ids:
            found["prompt"] = a or b
        answer = re.search(r"answer=(\d+) \(", out)
        if answer:
            found["answer"] = answer.group(1)
    other = player if who == "gm" else gm
    pushed = other.read(1.0).strip()
    if pushed:
        print(f"----- pushed to [{'player' if who == 'gm' else 'gm'}]")
        print(pushed)
        if who != "gm":
            ids = re.findall(r"GM prompt \[(\d+)\]", pushed)
            if ids:
                found["prompt"] = ids[-1]
time.sleep(1)
print("===== [player] late")
print(player.read(2).strip())
print("===== [gm] late")
print(gm.read(2).strip())
print("FOUND", json.dumps(found))
